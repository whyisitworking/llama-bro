cmake_minimum_required(VERSION 3.14)
project(actstat CXX C)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_SRC} llama)
add_executable(actstat actstat.cpp)
target_link_libraries(actstat PRIVATE llama-common llama)
target_include_directories(actstat PRIVATE ${LLAMA_SRC}/common)
target_compile_features(actstat PRIVATE cxx_std_17)
