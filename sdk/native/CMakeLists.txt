# Llama-Bro SDK native build (P1). Builds libllamabro (the C API over llama-server's server-context)
# from the patched llama.cpp copy made by scripts/prepare-llama.sh.

project(llamabro C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(LB_LLAMA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/build/llama.cpp" CACHE PATH "Patched llama.cpp copy (prepare-llama.sh)")
# not a cache entry: a build directory made before an upgrade must not keep the old name
set(LB_LLAMA_BUILD "b12479")
set(LB_VERSION "0.1.0" CACHE STRING "llamabro version")
option(LB_BUILD_TESTS "Build host tests, lb-run and llama-server" OFF)
# On by default for Android packaging; host builds keep upstream's names, so the contract test's llama-server
# finds its backends exactly as upstream does.
if(ANDROID)
    option(LB_PREFIX_LIBS "Give the shipped llama.cpp libraries a llamabro- prefix" ON)
else()
    option(LB_PREFIX_LIBS "Give the shipped llama.cpp libraries a llamabro- prefix" OFF)
endif()

if(NOT EXISTS "${LB_LLAMA_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "No llama.cpp copy at ${LB_LLAMA_DIR}: run sdk/native/scripts/prepare-llama.sh first")
endif()

# llama.cpp's cmake/build-info.cmake asks `git rev-parse --short HEAD` and `git rev-list --count HEAD` in the
# copy's directory. The copy isn't a repo, so git answers for whatever repo contains it (this one), and
# /props and every system_fingerprint would name the wrong build (F24). Point it at a stand-in that answers
# for the submodule: the commit from the copy's .lb-stamp and the build number from LB_LLAMA_BUILD.
file(STRINGS "${LB_LLAMA_DIR}/.lb-stamp" lb_stamp_commit REGEX "^commit ")
string(REGEX REPLACE "^commit ([0-9a-f]+).*" "\\1" lb_llama_commit "${lb_stamp_commit}")
string(SUBSTRING "${lb_llama_commit}" 0 7 lb_llama_commit_short)
string(REGEX REPLACE "^b" "" lb_llama_build_number "${LB_LLAMA_BUILD}")
if(NOT lb_llama_commit_short MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "no commit in ${LB_LLAMA_DIR}/.lb-stamp")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/lb-git-build-info.sh"
"#!/bin/sh
# Generated by sdk/native/CMakeLists.txt: answers llama.cpp build-info.cmake for the submodule (F24).
case \"$*\" in
  'rev-parse --short HEAD') echo ${lb_llama_commit_short} ;;
  'rev-list --count HEAD')  echo ${lb_llama_build_number} ;;
  *) exit 1 ;;
esac
")
file(CHMOD "${CMAKE_BINARY_DIR}/lb-git-build-info.sh" PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
set(GIT_EXECUTABLE "${CMAKE_BINARY_DIR}/lb-git-build-info.sh" CACHE FILEPATH "" FORCE)

# Host JVM: the same JNI layer, so Kotlin's JniEngine runs against the real core in JVM tests (host end-to-end,
# sdk/scripts/check.sh). The JDK bundled with Android Studio has no headers; the NDK's jni.h is self-contained and
# follows the standard JNI function table, so a copy of it is used.
option(LB_HOST_JNI "Build the JNI layer into the host libllamabro (needs the NDK's jni.h)" OFF)
if(LB_HOST_JNI AND NOT ANDROID)
    file(GLOB lb_ndk_jni "$ENV{ANDROID_HOME}/ndk/29.0.14206865/toolchains/llvm/prebuilt/*/sysroot/usr/include/jni.h")
    if(NOT lb_ndk_jni)
        message(FATAL_ERROR "LB_HOST_JNI: jni.h not found under $ENV{ANDROID_HOME}/ndk/29.0.14206865 (set ANDROID_HOME)")
    endif()
    list(GET lb_ndk_jni 0 lb_ndk_jni)
    configure_file("${lb_ndk_jni}" "${CMAKE_BINARY_DIR}/jni-include/jni.h" COPYONLY)
    target_include_directories(llamabro PRIVATE "${CMAKE_BINARY_DIR}/jni-include")
    target_sources(llamabro PRIVATE jni/lb_jni.cpp)
endif()

if(ANDROID)
    target_sources(llamabro PRIVATE jni/lb_jni.cpp)
    target_link_libraries(llamabro PRIVATE log)
    # export the C API and JNI_OnLoad only; server-context, linked in statically, stays private
    target_link_options(llamabro PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/bridge/exports.map")
    set_target_properties(llamabro PROPERTIES LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/bridge/exports.map")
    # 16 KB pages (Android 15+): NDK r28+ aligns arm64 by default; stated here so it can't regress
    target_link_options(llamabro PRIVATE "-Wl,-z,max-page-size=16384")
endif()
target_compile_definitions(llamabro PRIVATE LB_VERSION="${LB_VERSION}" LB_LLAMA_BUILD="${LB_LLAMA_BUILD}")

# EXCLUDE_FROM_ALL hides the CPU backend modules: nothing links them, they are loaded at runtime.
get_property(lb_ggml_targets DIRECTORY ${LB_LLAMA_DIR}/ggml/src PROPERTY BUILDSYSTEM_TARGETS)
set(LB_CPU_BACKENDS "")
foreach(t ${lb_ggml_targets})
    if(t MATCHES "^ggml-cpu")
        list(APPEND LB_CPU_BACKENDS ${t})
    endif()
endforeach()
if(NOT LB_CPU_BACKENDS)
    message(FATAL_ERROR "no ggml-cpu backend targets found")
endif()
message(STATUS "llamabro: CPU backends ${LB_CPU_BACKENDS}")
add_dependencies(llamabro ${LB_CPU_BACKENDS})

# Our own library names, so an app that also ships another llama.cpp build can't get a duplicate-file error or
# load the wrong libllama.so (findings F25). OUTPUT_NAME only renames the files; nothing in llama.cpp changes.
foreach(t ggml-base ggml llama llama-common mtmd ggml-hexagon ggml-opencl ggml-vulkan)
    if(LB_PREFIX_LIBS AND TARGET ${t})
        set_target_properties(${t} PROPERTIES OUTPUT_NAME "llamabro-${t}")
    endif()
endforeach()
# The bridge loads the CPU variants by file name (runtime.cpp): CMake hands it the list.
set(lb_cpu_files "")
foreach(t ${LB_CPU_BACKENDS})
    get_target_property(t_type ${t} TYPE)
    if(t_type STREQUAL "MODULE_LIBRARY" OR t_type STREQUAL "SHARED_LIBRARY")
        if(LB_PREFIX_LIBS)
            set_target_properties(${t} PROPERTIES OUTPUT_NAME "llamabro-${t}")
        endif()
        list(APPEND lb_cpu_files "$<TARGET_FILE_NAME:${t}>")
    endif()
endforeach()
list(JOIN lb_cpu_files "," lb_cpu_files_joined)
target_compile_definitions(llamabro PRIVATE "LB_CPU_BACKEND_FILES=\"${lb_cpu_files_joined}\"")
